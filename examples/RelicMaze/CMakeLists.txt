cmake_minimum_required(VERSION 3.16)
project(RelicMaze C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_library(relic_maze_core STATIC core/relic_maze.c)
target_include_directories(relic_maze_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/core)
add_executable(relic_maze_test tests/test_game.c)
target_link_libraries(relic_maze_test PRIVATE relic_maze_core)
enable_testing()
add_test(NAME relic_maze_logic COMMAND relic_maze_test)

add_executable(relic_maze_board_test tests/test_board_texture.c ios/board_texture.c)
target_include_directories(relic_maze_board_test PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/ios)
target_link_libraries(relic_maze_board_test PRIVATE relic_maze_core)
add_test(NAME relic_maze_board_texture COMMAND relic_maze_board_test)

find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME relic_maze_level_solver COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/verify_level.py)
add_test(NAME relic_maze_armv6_bundle COMMAND ${Python3_EXECUTABLE} -m unittest discover -s ${CMAKE_CURRENT_SOURCE_DIR}/tests -p "test_*.py")
add_test(NAME relic_maze_browser COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/run_browser_test.py)
set_tests_properties(relic_maze_level_solver relic_maze_armv6_bundle relic_maze_browser PROPERTIES ENVIRONMENT "PYTHONDONTWRITEBYTECODE=1")
